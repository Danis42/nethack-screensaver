# Threat Model Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The bot's three danger predicates (`is_dangerous_monster`, `is_monster_faster`, `imminent_death_on_melee`) use NetHack's real monster speed, level and attack dice instead of name lists and fixed HP limits.

**Architecture:** A generated data module holds every monster's attacks (from NetHack 3.6.6 `monst.c`, aligned to NLE's monster indices). A pure `threat.py` turns attacks, level and speed into expected damage per turn and "special attack" flags. `monster_utils.py` rewires the three predicates to it, keeping their names and signatures so all callers stay unchanged.

**Tech Stack:** Python 3.13, NLE 1.3.0 (`nle.nethack.permonst`, `glyph_to_mon`), pytest, `uv`.

**Spec:** `docs/superpowers/specs/2026-10-21-threat-model-design.md`

## Global Constraints

- Predicate names and signatures stay: `is_monster_faster(agent, monster)`, `is_dangerous_monster(monster)`, `imminent_death_on_melee(agent, monster)`; `monster` is the bot's tuple `(distance, y, x, permonst, glyph)`.
- `threat.py` is pure (numbers in, numbers out) and imports only `monster_data`.
- `monster_data.py` is generated by `scripts/gen_monster_data.py` and never edited by hand; its order equals NLE's monster index (`nle.nethack.permonst(i)`, `i < nle.nethack.NUMMONS`).
- Thresholds, verbatim: dangerous when `damage_per_turn` against AC 10 and speed 12 is ≥ 4, or a special attack, or a pet name (as today); imminent death when HP ≤ max(8, 2 × `damage_per_turn` against our AC and speed); our speed 12, or 16 for Archeologist, Monk, Samurai.
- Unknown monster index: no attacks, not special, speed from `mmove`.
- Whole suite before and after: `uv run pytest -q -m ""`.
- Branch `threat-model`; never push. Merging waits for the bench experiment.

## Review Focus

- Pets: their glyphs are pet glyphs, not monster glyphs; the index helper must map them (`glyph_to_pet`) — tested in Task 3.
- A monster with negative-AC math: our AC below 0 must lower the hit chance, never raise it above 1 or below 0 — tested in Task 2.
- Weapon-attack monsters (dwarves, orcs) get the monst.c dice only (no weapon bonus); that is accepted and documented in `threat.py` — checked by a Task 2 test value for a dwarf.
- `is_dangerous_monster` gets no agent; it must not touch agent state — tested in Task 3 with only a monster tuple.
- The generator must fail loudly on a name mismatch rather than writing a misaligned table — tested in Task 1.

# scripts/gen_monster_data.py
"""Write autoascend/combat/monster_data.py from NetHack 3.6.6 src/monst.c.

NLE's permonst has level, speed and AC but no attacks; the bot's threat model needs them.
Usage: uv run python scripts/gen_monster_data.py path/to/monst.c
(https://raw.githubusercontent.com/NetHack/NetHack/NetHack-3.6.6_Released/src/monst.c)"""
import re
import sys
from pathlib import Path

OUT = Path(__file__).resolve().parent.parent / "autoascend" / "combat" / "monster_data.py"

# include/monattk.h, NetHack 3.6.6
AT = {"AT_NONE": 0, "AT_CLAW": 1, "AT_BITE": 2, "AT_KICK": 3, "AT_BUTT": 4, "AT_TUCH": 5, "AT_STNG": 6,
      "AT_HUGS": 7, "AT_SPIT": 10, "AT_ENGL": 11, "AT_BREA": 12, "AT_EXPL": 13, "AT_BOOM": 14, "AT_GAZE": 15,
      "AT_TENT": 16, "AT_WEAP": 254, "AT_MAGC": 255}
AD = {"AD_PHYS": 0, "AD_MAGM": 1, "AD_FIRE": 2, "AD_COLD": 3, "AD_SLEE": 4, "AD_DISN": 5, "AD_ELEC": 6,
      "AD_DRST": 7, "AD_ACID": 8, "AD_SPC1": 9, "AD_SPC2": 10, "AD_BLND": 11, "AD_STUN": 12, "AD_SLOW": 13,
      "AD_PLYS": 14, "AD_DRLI": 15, "AD_DREN": 16, "AD_LEGS": 17, "AD_STON": 18, "AD_STCK": 19, "AD_SGLD": 20,
      "AD_SITM": 21, "AD_SEDU": 22, "AD_TLPT": 23, "AD_RUST": 24, "AD_CONF": 25, "AD_DGST": 26, "AD_HEAL": 27,
      "AD_WRAP": 28, "AD_WERE": 29, "AD_DRDX": 30, "AD_DRCO": 31, "AD_DRIN": 32, "AD_DISE": 33, "AD_DCAY": 34,
      "AD_SSEX": 35, "AD_HALU": 36, "AD_DETH": 37, "AD_PEST": 38, "AD_FAMN": 39, "AD_SLIM": 40, "AD_ENCH": 41,
      "AD_CORR": 42, "AD_CLRC": 240, "AD_SPEL": 241, "AD_RBRE": 242, "AD_SAMU": 252, "AD_CURS": 253}
ATTK = re.compile(r"ATTK\((\w+),\s*(\w+),\s*(\d+),\s*(\d+)\)")
DISABLED = re.compile(r"^#if 0\b.*?^#endif[^\n]*$", re.S | re.M)


def parse(text):
    """[(name, attacks)] for every MON(...) entry outside `#if 0` blocks, in file order."""
    out = []
    for chunk in DISABLED.sub("", text).split('MON("')[1:]:
        name = chunk[:chunk.index('"')]
        body = chunk[:chunk.index("SIZ(")]
        out.append((name, tuple((AT[a], AD[d], int(n), int(s)) for a, d, n, s in ATTK.findall(body))))
    return out


def align(parsed, names):
    """The parsed entries in the order of `names` (NLE's monster indices); entries NLE lacks are skipped."""
    out, j = [], 0
    for i, name in enumerate(names):
        while j < len(parsed) and parsed[j][0] != name:
            j += 1
        if j == len(parsed):
            raise SystemExit(f"monster {i} {name!r} not found in monst.c order")
        out.append(parsed[j])
        j += 1
    return out


def main(path):
    import nle.nethack as nh
    names = [nh.permonst(i).mname for i in range(nh.NUMMONS)]
    table = align(parse(Path(path).read_text()), names)
    lines = ['"""Generated by scripts/gen_monster_data.py from NetHack 3.6.6 src/monst.c; do not edit.',
             "",
             "Monster data from NetHack, distributed under the NetHack General Public License.",
             'Index = NLE monster index; each entry: (name, ((aatyp, adtyp, damn, damd), ...))."""',
             "MONSTERS = ("]
    lines += [f"    ({name!r}, {attacks!r})," for name, attacks in table]
    lines.append(")")
    OUT.write_text("\n".join(lines) + "\n")
    print(f"wrote {OUT} ({len(table)} monsters)")


if __name__ == "__main__":
    main(sys.argv[1])
```

# autoascend/combat/threat.py
"""How dangerous a monster is, from NetHack's own numbers (sub-project 1, docs/ROADMAP.md).

Pure functions over the attack table in monster_data.py. A melee attack hits when
10 + our AC + monster level > d20 (mattacku); negative AC counts as -rnd(-AC). Weapon
users get their monst.c dice only, without the weapon's own damage."""
from .monster_data import MONSTERS

BASE_SPEED = 12
FAST_SPEED = 16  # intrinsic Fast: about one extra move in three
DANGER_HP_PER_TURN = 4  # against AC 10 and speed 12

# include/monattk.h
MELEE = frozenset({1, 2, 3, 4, 5, 6, 7, 11, 16, 254})  # claw bite kick butt touch sting hug engulf tentacle weapon
SPECIAL = frozenset({
    7, 30, 31,  # poison: drain Str, Dex, Con
    14,  # paralysis
    15,  # level drain
    18,  # stoning
    26,  # digestion
    29,  # lycanthropy
    32,  # brain eating
    33,  # disease
    40,  # sliming
})


def attacks(index):
    return MONSTERS[index][1] if 0 <= index < len(MONSTERS) else ()


def hit_chance(mlevel, our_ac):
    ac = our_ac if our_ac >= 0 else (our_ac - 1) / 2
    return min(max((10 + ac + mlevel - 1) / 20, 0.0), 1.0)


def damage_per_turn(index, mlevel, mspeed, our_ac, our_speed=BASE_SPEED):
    """Expected HP this monster takes from us per turn of ours in melee."""
    hit = hit_chance(mlevel, our_ac)
    per_move = sum(hit * n * (d + 1) / 2 for at, ad, n, d in attacks(index) if at in MELEE)
    return per_move * mspeed / our_speed


def has_special_attack(index):
    """An attack that ends runs in ways HP does not show (poison, lycanthropy, stoning, ...)."""
    return any(ad in SPECIAL for at, ad, n, d in attacks(index))
```

```python
FAST_ROLES = (0, 5, 9)  # Character.ARCHEOLOGIST, MONK, SAMURAI: intrinsically Fast from turn 1
PET_NAMES = ('dog', 'cat', 'kitten', 'pony', 'horse')


def monster_index(glyph):
    """NLE monster index of a monster or pet glyph, -1 otherwise."""
    if nh.glyph_is_monster(glyph):
        return nh.glyph_to_mon(glyph)
    if nh.glyph_is_pet(glyph):
        return nh.glyph_to_pet(glyph)
    return -1


def our_speed(agent):
    return threat.FAST_SPEED if agent.character.role in FAST_ROLES else threat.BASE_SPEED


def is_monster_faster(agent, monster):
    _, y, x, mon, _ = monster
    return mon.mmove > our_speed(agent)


def imminent_death_on_melee(agent, monster):
    _, y, x, mon, glyph = monster
    damage = threat.damage_per_turn(monster_index(glyph), mon.mlevel, mon.mmove,
                                    agent.blstats.armor_class, our_speed(agent))
    return agent.blstats.hitpoints <= max(8, 2 * damage)


def is_dangerous_monster(monster):
    _, y, x, mon, glyph = monster
    if any(name in mon.mname for name in PET_NAMES):
        return True
    index = monster_index(glyph)
    return threat.has_special_attack(index) or \
        threat.damage_per_turn(index, mon.mlevel, mon.mmove, 10) >= threat.DANGER_HP_PER_TURN
```
